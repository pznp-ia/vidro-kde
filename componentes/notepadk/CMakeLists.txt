cmake_minimum_required(VERSION 3.20)
project(vidrofocus LANGUAGES CXX)
find_package(ECM REQUIRED NO_MODULE)
set(CMAKE_MODULE_PATH ${ECM_MODULE_PATH})
include(KDEInstallDirs)
include(KDECMakeSettings)
include(KDECompilerSettings NO_POLICY_SCOPE)
find_package(Qt6 REQUIRED COMPONENTS Core Widgets DBus)
find_package(KF6SyntaxHighlighting REQUIRED)
find_package(KF6 REQUIRED COMPONENTS TextEditor CoreAddons XmlGui I18n)
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin)
kcoreaddons_add_plugin(vidrofocusplugin SOURCES plugin.cpp INSTALL_NAMESPACE "kf6/ktexteditor")
target_link_libraries(vidrofocusplugin PRIVATE KF6::TextEditor KF6::CoreAddons KF6::XmlGui KF6::I18n KF6::SyntaxHighlighting Qt6::Widgets Qt6::DBus)
